import { Image } from 'astro:assets';
import { speakers } from '~/data/speakers2026';

{speakers.map((speaker) => (
  <section id={speaker.id}>
    <h3>{speaker.name}</h3>
    <h4>
      {speaker.role}
      {speaker.companyUrl && (
        <> | <a href={speaker.companyUrl}>{speaker.company}</a></>
      )}
    </h4>
    {!speaker.companyUrl && <p>{speaker.company}</p>}

    {speaker.image && (
      <Image src={speaker.image} width={300} alt={speaker.imageAlt ?? speaker.name} format="webp" quality={80} />
    )}

    <p>{speaker.talkTitle}</p>
    {speaker.talkDescription.map((paragraph) => <p>{paragraph}</p>)}

    <ul>
      {speaker.links.map((link) => <li><a href={link.href}>{link.label}</a></li>)}
    </ul>
  </section>
))}
